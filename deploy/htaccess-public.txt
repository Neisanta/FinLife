# FinLife — front controller (dentro de /public)
# Suba este arquivo em public_html/public e renomeie para ".htaccess"
Options -Indexes

<IfModule mod_rewrite.c>
    RewriteEngine On

    # Envia tudo que não for arquivo/diretório para o index.php
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteRule ^ index.php [QSA,L]
</IfModule>
